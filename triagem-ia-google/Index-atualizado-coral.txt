    <div id="root" class="loading">Carregando avaliações...</div>
    <script>
      const state = { candidates: [], dashboard: null, selected: null, view: "overview" };
      const esc = value => String(value ?? "").replace(/[&<>\"]/g, char => ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;"})[char]);
      const formatScore = value => value === null || value === undefined ? "—" : `${Math.round(value)}`;
      const recommendationText = value => ({ RECOMENDADO: "Recomendado", PARCIALMENTE_RECOMENDADO: "Parcialmente recomendado", NAO_RECOMENDADO: "Não recomendado" })[value] || "Não recomendado";
      const chip = value => `<span class="chip ${esc(value)}">${esc(recommendationText(value))}</span>`;
      const percent = count => state.dashboard?.total ? Math.round((count / state.dashboard.total) * 100) : 0;
      function shell(content) { return `<div class="app"><aside class="sidebar"><div class="brand"><div class="brand-mark">IA</div><div><strong>Triagem IA</strong><small>Talentos e decisões</small></div></div><div class="nav-label">Navegação</div><nav class="nav"><button class="${state.view === "overview" ? "active" : ""}" onclick="showView('overview')">Visão geral</button><button class="${state.view === "candidates" ? "active" : ""}" onclick="showView('candidates')">Candidatos</button></nav><div class="account">Acesso autorizado<br><strong>${esc(state.user?.email || "")}</strong></div></aside><main class="main">${content}</main></div>`; }
      function overview() { const d = state.dashboard; return `<section><div class="topline"><div><div class="eyebrow">Inteligência de talentos</div><h1>Decisões mais claras.<br>Contratações melhores.</h1><p class="subtitle">Acompanhe os currículos processados pela IA, identifique aderência e navegue pelos resultados em um só lugar.</p></div><div class="status">Dados atualizados da planilha</div></div><div class="metrics"><div class="metric"><label>Candidatos avaliados</label><b>${d.total}</b></div><div class="metric"><label>Pontuação média</label><b class="${d.average === null ? "dash" : ""}">${formatScore(d.average)}</b></div><div class="metric"><label>Recomendados</label><b>${d.recommended}</b></div><div class="metric"><label>Não recomendados</label><b>${d.notRecommended}</b></div></div><section class="panel"><div class="panel-header"><div><div class="eyebrow">Distribuição</div><h2>Decisões da IA</h2></div><span class="chip CORAL_TOTAL">Total: ${d.total}</span></div><div class="distribution">${d.distribution.map(item => `<div class="dist-line"><div><b>${esc(item.label)}</b><small>${esc(item.description)}</small></div><div class="bar"><span style="width:${percent(item.count)}%"></span></div><em>${item.count} · ${percent(item.count)}%</em></div>`).join("")}</div></section></section>`; }
      function candidates() { return `<section><div class="topline"><div><div class="eyebrow">Base de talentos</div><h1>Candidatos avaliados.</h1><p class="subtitle">Pesquise perfis, compare recomendações e consulte a análise completa de cada candidato.</p></div></div><section class="panel"><div class="filters"><input id="search" oninput="renderCandidateList()" placeholder="Pesquisar por nome ou e-mail" aria-label="Pesquisar candidato"><select id="recommendation" onchange="renderCandidateList()" aria-label="Filtrar recomendação"><option value="">Todas as recomendações</option><option value="RECOMENDADO">Recomendados</option><option value="PARCIALMENTE_RECOMENDADO">Parcialmente recomendados</option><option value="NAO_RECOMENDADO">Não recomendados</option></select><select id="score" onchange="renderCandidateList()" aria-label="Filtrar pontuação"><option value="">Todas as pontuações</option><option value="80">80 a 100</option><option value="60">60 a 79</option><option value="0">Abaixo de 60</option></select></div><div id="candidate-list" class="candidate-list" style="margin-top:18px"></div></section></section>`; }
      function renderCandidateList() { const search = (document.getElementById("search")?.value || "").toLowerCase(); const recommendation = document.getElementById("recommendation")?.value || ""; const scoreBand = document.getElementById("score")?.value || ""; const matches = state.candidates.filter(item => { const text = `${item.name} ${item.email}`.toLowerCase(); const score = item.totalScore ?? -1; const inBand = !scoreBand || (scoreBand === "80" && score >= 80) || (scoreBand === "60" && score >= 60 && score < 80) || (scoreBand === "0" && score < 60); return text.includes(search) && (!recommendation || item.recommendation === recommendation) && inBand; }); const target = document.getElementById("candidate-list"); target.innerHTML = matches.length ? matches.map(item => `<button class="candidate" onclick="openCandidate('${esc(item.id)}')"><span><span class="candidate-name">${esc(item.name)}</span><span class="candidate-meta">${esc(item.email || item.submittedAt || "Sem contato informado")}</span></span><span class="score">${formatScore(item.totalScore)}</span>${chip(item.recommendation)}</button>`).join("") : `<div class="empty">Nenhum candidato foi encontrado com estes filtros.</div>`; }
      function detail() { const c = state.selected; const criterion = (title, score, limit, text) => `<article class="criterion"><span>${title}</span><b>${formatScore(score)} <small>/ ${limit}</small></b><p>${esc(text || "Justificativa não informada.")}</p></article>`; const words = (items, missing) => items.length ? items.map(item => `<span class="keyword ${missing ? "missing" : ""}">${esc(item)}</span>`).join("") : `<span class="keyword ${missing ? "missing" : ""}">Nenhuma informada</span>`; return `<section><button class="back" onclick="showView('candidates')">← Voltar para candidatos</button><section class="panel"><div class="detail-hero"><div><div class="eyebrow">Perfil avaliado</div><h2>${esc(c.name)}</h2>${chip(c.recommendation)}<p class="detail-contact">${esc(c.email || "E-mail não informado")}<br>${esc(c.phone || "Telefone não informado")}<br>${esc(c.submittedAt || "Data não informada")}</p>${c.resumeUrl ? `<a class="resume" href="${esc(c.resumeUrl)}" target="_blank" rel="noopener">Abrir currículo no Google Drive</a>` : ""}</div><div><div class="eyebrow">Pontuação total</div><div class="detail-score">${formatScore(c.totalScore)}</div></div></div><div class="criteria">${criterion("Formação", c.scores.education, 25, c.justifications.education)}${criterion("Experiência", c.scores.experience, 35, c.justifications.experience)}${criterion("Habilidades", c.scores.skills, 25, c.justifications.skills)}${criterion("Idiomas e certificações", c.scores.languages, 15, c.justifications.languages)}</div></section><section class="panel"><div class="eyebrow">Palavras-chave</div><h2>Indicadores de aderência</h2><p class="subtitle" style="margin:10px 0 8px">Encontradas</p><div class="keyword-row">${words(c.foundKeywords, false)}</div><p class="subtitle" style="margin:18px 0 8px">A desenvolver</p><div class="keyword-row">${words(c.missingKeywords, true)}</div></section><section class="panel"><div class="eyebrow">Leitura final</div><h2>Análise da IA</h2><p class="analysis">${esc(c.analysisDetail || "Análise detalhada não informada.")}</p></section></section>`; }
      function render() { document.getElementById("root").innerHTML = shell(state.view === "overview" ? overview() : state.view === "candidates" ? candidates() : detail()); if (state.view === "candidates") renderCandidateList(); }
      function applyInitialData(data) { state.user = data.user; state.dashboard = data.dashboard; state.candidates = data.candidates; render(); }
      function refreshData() { document.getElementById("root").innerHTML = '<div class="loading">Atualizando avaliações...</div>'; google.script.run.withSuccessHandler(applyInitialData).withFailureHandler(showError).getInitialData(); }
      function showView(view) { state.view = view; if (view === "overview") refreshData(); else render(); }
      function openCandidate(id) { google.script.run.withSuccessHandler(candidate => { state.selected = candidate; state.view = "detail"; render(); }).withFailureHandler(showError).getCandidate(id); }
      function showError(error) { document.getElementById("root").innerHTML = `<div class="error"><div class="eyebrow">Acesso indisponível</div><h2>Não foi possível abrir o painel.</h2><p>${esc(error?.message || String(error))}</p>`; }
      refreshData();
    </script>
